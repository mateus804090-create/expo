import { View, Text, TextInput, TouchableOpacity, ScrollView, StyleSheet, Alert, Linking } from "react-native";
import AsyncStorage from "@react-native-async-storage/async-storage";

export default function App() {
  const [tela, setTela] = useState("dashboard");
  const [clientes, setClientes] = useState([]);
  const [clienteAtual, setClienteAtual] = useState(null);

  // Campos de Cadastro / Edição
  const [nome, setNome] = useState("");
  const [telefone, setTelefone] = useState("");
  const [valor, setValor] = useState("");
  const [parcelas, setParcelas] = useState("");
  const [frequencia, setFrequencia] = useState("diario"); // "diario" ou "semanal"
  const [dataPrimeiroVencimento, setDataPrimeiroVencimento] = useState("");

  // Filtro de Busca
  const [pesquisa, setPesquisa] = useState("");

  // Campos de Movimentação
  const [valorPagamento, setValorPagamento] = useState("");
  const [valorCaixa, setValorCaixa] = useState("");
  const [comprovante, setComprovante] = useState("");

  // Caixa e Métricas
  const [caixa, setCaixa] = useState(0);
  const [entradaDia, setEntradaDia] = useState(0);

  useEffect(() => {
    carregarDados();
    carregarCaixa();
    setDataPrimeiroVencimento(new Date().toLocaleDateString("pt-BR"));
  }, []);

  // ================= DADOS =================

  async function carregarDados() {
    const c = await AsyncStorage.getItem("clientes_parcelas");
    if (c) setClientes(JSON.parse(c));
  }

  async function salvar(lista) {
    await AsyncStorage.setItem("clientes_parcelas", JSON.stringify(lista));
    setClientes(lista);
  }

  // ================= CAIXA =================

  async function carregarCaixa() {
    const caixaSalvo = await AsyncStorage.getItem("caixa_valor");
    setCaixa(Number(caixaSalvo) || 0);
  }

  async function atualizarEntrada(valor) {
    const novaEntrada = entradaDia + valor;
    const novoCaixa = caixa + valor;

    setEntradaDia(novaEntrada);
    setCaixa(novoCaixa);

    await AsyncStorage.setItem("caixa_valor", String(novoCaixa));
  }

  async function ajustarCaixa() {
    const valorLimpo = valorCaixa.replace(",", ".");
    if (valorLimpo === "" || isNaN(Number(valorLimpo))) {
      Alert.alert("Erro", "Digite um valor válido para o ajuste.");
      return;
    }

    const novoCaixa = (Number(caixa) || 0) + Number(valorLimpo);
    setCaixa(novoCaixa);
    await AsyncStorage.setItem("caixa_valor", String(novoCaixa));
    setValorCaixa("");
    Alert.alert("Sucesso", "Caixa ajustado!");
  }

  // ================= AUXILIAR: GERA PARCELAS POR FREQUÊNCIA =================
  function gerarListaParcelas(qtd, total, dataInicio, freq) {
    const valorParcela = total / qtd;
    const lista = [];

    const partes = dataInicio.split("/");
    if (partes.length !== 3) return [];
    
    // Cria o objeto de data baseado no input DD/MM/AAAA
    let dataBase = new Date(partes[2], partes[1] - 1, partes[0]);

    if (freq === "diario") {
      // Regra Diária: Pula domingo (0)
      if (dataBase.getDay() === 0) {
        dataBase.setDate(dataBase.getDate() + 1); // Se for domingo, joga pra segunda
      }
      
      let dataAtual = new Date(dataBase);

      for (let i = 1; i <= qtd; i++) {
        lista.push({
          numero: i,
          valor: valorParcela,
          pago: 0,
          dataPagamento: null,
          dataVencimento: dataAtual.toLocaleDateString("pt-BR")
        });
        
        // Avança 1 dia para a próxima parcela
        dataAtual.setDate(dataAtual.getDate() + 1);
        
        // Se o próximo dia for domingo, pula pra segunda
        if (dataAtual.getDay() === 0) {
          dataAtual.setDate(dataAtual.getDate() + 1);
        }
      }
    } else if (freq === "semanal") {
      // Regra Semanal: Travar sempre no Sábado (6)
      let dataAtual = new Date(dataBase);
      let diaSemana = dataAtual.getDay();
      
      if (diaSemana !== 6) { 
        let diasParaSabado = 6 - diaSemana;
        dataAtual.setDate(dataAtual.getDate() + diasParaSabado);
      }

      for (let i = 1; i <= qtd; i++) {
        lista.push({
          numero: i,
          valor: valorParcela,
          pago: 0,
          dataPagamento: null,
          dataVencimento: dataAtual.toLocaleDateString("pt-BR")
        });
        
        // Avança exatos 7 dias para o próximo sábado
        dataAtual.setDate(dataAtual.getDate() + 7);
      }
    }
    
    return lista;
  }

  // ================= CLIENTE: ADICIONAR =================

  function adicionarCliente() {
    const valorLimpo = valor.replace(",", ".");

    if (!nome || !valorLimpo || !parcelas || !dataPrimeiroVencimento) {
      Alert.alert("Atenção", "Preencha todos os campos obrigatórios.");
      return;
    }

    const total = Number(valorLimpo);
    const qtd = Number(parcelas);
    
    const listaParcelas = gerarListaParcelas(qtd, total, dataPrimeiroVencimento, frequencia);
    if (listaParcelas.length === 0) {
      Alert.alert("Erro", "Formato de data inválido. Use DD/MM/AAAA");
      return;
    }

    const novo = {
      id: Date.now().toString(),
      nome,
      telefone,
      valorTotal: total,
      pagoTotal: 0,
      parcelas: qtd,
      frequencia,
      parcelasLista: listaParcelas,
    };

    salvar([...clientes, novo]);
    
    setNome(""); setTelefone(""); setValor(""); setParcelas(""); 
    setFrequencia("diario"); setDataPrimeiroVencimento(new Date().toLocaleDateString("pt-BR"));
    setTela("dashboard");
  }

  // ================= CLIENTE: EDITAR CADASTRO =================

  function abrirEditar(cliente) {
    setNome(cliente.nome);
    setTelefone(cliente.telefone);
    setValor(String(cliente.valorTotal));
    setParcelas(String(cliente.parcelas));
    setFrequencia(cliente.frequencia || "diario");
    if (cliente.parcelasLista && cliente.parcelasLista[0]) {
      setDataPrimeiroVencimento(cliente.parcelasLista[0].dataVencimento);
    } else {
      setDataPrimeiroVencimento(new Date().toLocaleDateString("pt-BR"));
    }
    setTela("editar");
  }

  function salvarEdicao() {
    const valorLimpo = valor.replace(",", ".");
    if (!nome || !valorLimpo || !parcelas || !dataPrimeiroVencimento) {
      Alert.alert("Atenção", "Preencha todos os campos.");
      return;
    }

    const total = Number(valorLimpo);
    const qtd = Number(parcelas);

    const listaNovaParcelas = gerarListaParcelas(qtd, total, dataPrimeiroVencimento, frequencia);
    if (listaNovaParcelas.length === 0) {
      Alert.alert("Erro", "Formato de data inválido. Use DD/MM/AAAA");
      return;
    }

    const listaAtualizada = clientes.map(c => {
      if (c.id === clienteAtual.id) {
        return {
          ...c,
          nome,
          telefone,
          valorTotal: total,
          parcelas: qtd,
          frequencia,
          pagoTotal: 0,
          parcelasLista: listaNovaParcelas
        };
      }
      return c;
    });

    salvar(listaAtualizada);
    
    setNome(""); setTelefone(""); setValor(""); setParcelas("");
    
    const atualizado = listaAtualizada.find(x => x.id === clienteAtual.id);
    setClienteAtual(atualizado);
    
    setTela("cliente");
    Alert.alert("Sucesso", "Cadastro atualizado e parcelas recalculadas!");
  }

  // ================= COMPROVANTE =================

  function gerarComprovante(cliente, valorPago) {
    const restante = (cliente.valorTotal - cliente.pagoTotal).toFixed(2);
    const data = new Date().toLocaleDateString();
    const parcelasPagas = cliente.parcelasLista.filter((p) => p.pago >= p.valor).length;
    const quitado = parcelasPagas === cliente.parcelas;

    const mensagemFinal = quitado 
      ? "\n🎉 Seu crédito foi finalizado com sucesso!" 
      : "\nObrigado pelo pagamento!";

    const textoComprovante = `🧾 *COMPROVANTE DE PAGAMENTO*
----------------------------------
*Cliente:* ${cliente.nome}
*Telefone:* ${cliente.telefone || "Não informado"}

*Parcelas Quitadas:* ${parcelasPagas}/${cliente.parcelas}
*Valor Pago Agora:* R$ ${Number(valorPago).toFixed(2)}
*Saldo Devedor:* R$ ${restante}

*Data:* ${data}
----------------------------------${mensagemFinal}

Seven Crédito`;

    setComprovante(textoComprovante);
  }

  // ================= COBRANÇA =================

  function enviarCobranca(cliente) {
    if (!cliente?.telefone) {
      Alert.alert("Erro", "Este cliente não possui telefone cadastrado.");
      return;
    }

    const hora = new Date().getHours();
    let saudacao = "Olá!";
    if (hora >= 5 && hora < 12) saudacao = "Bom dia!";
    else if (hora >= 12 && hora < 18) saudacao = "Boa tarde!";
    else saudacao = "Boa noite!";

    const textoCobranca = `${saudacao} Tudo bem? Passando para lembrar sobre as parcelas do seu crédito com a *Seven Crédito*.

Estamos no aguardo do pagamento para atualizar o seu cadastro. 👍

Caso já tenha efetuado o pagamento, por favor desconsidere ou envie o comprovante. Obrigado!`;

    const tel = cliente.telefone.replace(/\D/g, "");
    const texto = encodeURIComponent(textoCobranca);

    Linking.openURL(`https://wa.me/55${tel}?text=${texto}`).catch(() => {
      Alert.alert("Erro", "Não foi possível abrir o WhatsApp.");
    });
  }

  // ================= WHATS COMPROVANTE =================

  function enviarWhatsApp() {
    if (!clienteAtual?.telefone) {
      Alert.alert("Erro", "Este cliente não possui telefone cadastrado.");
      return;
    }
    const tel = clienteAtual.telefone.replace(/\D/g, "");
    const texto = encodeURIComponent(comprovante);

    Linking.openURL(`https://wa.me/55${tel}?text=${texto}`).catch(() => {
      Alert.alert("Erro", "Não foi possível abrir o WhatsApp.");
    });
  }

  // ================= PAGAMENTO =================

  function pagarManual(index, valor) {
    const valorLimpo = valor.replace(",", ".");
    if (valorLimpo === "" || isNaN(Number(valorLimpo)) || Number(valorLimpo) <= 0) {
      Alert.alert("Erro", "Digite um valor de pagamento válido.");
      return;
    }

    const lista = [...clientes];
    const c = lista[index];

    let restante = Number(valorLimpo);
    let aplicado = 0;

    for (let p of c.parcelasLista) {
      if (restante <= 0) break;
      const falta = p.valor - p.pago;

      if (falta > 0) {
        const pagar = Math.min(falta, restante);
        p.pago += pagar;
        restante -= pagar;
        aplicado += pagar;

        if (p.pago >= p.valor) {
          p.dataPagamento = new Date().toLocaleDateString();
        }
      }
    }

    c.pagoTotal += aplicado;

    gerarComprovante(c, aplicado);
    atualizarEntrada(aplicado);
    salvar(lista);
    
    setClienteAtual({ ...c }); 
    Alert.alert("Sucesso", `Pagamento de R$ ${aplicado.toFixed(2)} registrado!`);
  }

  // ================= REMOVER PAGAMENTO =================

  function removerUltimoPagamento(index) {
    const lista = [...clientes];
    const c = lista[index];
    let valorRemovido = 0;

    for (let i = c.parcelasLista.length - 1; i >= 0; i--) {
      if (c.parcelasLista[i].pago > 0) {
        valorRemovido = c.parcelasLista[i].pago;
        c.pagoTotal -= valorRemovido;
        c.parcelasLista[i].pago = 0;
        c.parcelasLista[i].dataPagamento = null;
        break;
      }
    }

    if (valorRemovido > 0) {
      const novoCaixa = caixa - valorRemovido;
      setCaixa(novoCaixa);
      AsyncStorage.setItem("caixa_valor", String(novoCaixa));
      
      salvar(lista);
      setClienteAtual({ ...c });
      setComprovante("");
      Alert.alert("Removido", "Último pagamento desfeito.");
    } else {
      Alert.alert("Aviso", "Nenhum pagamento encontrado.");
    }
  }

  // ================= EXCLUIR =================

  function excluirCliente(id) {
    Alert.alert("Excluir Cliente", "Tem certeza que deseja apagar este cliente?", [
      { text: "Cancelar", style: "cancel" },
      {
        text: "Excluir",
        style: "destructive",
        onPress: () => {
          salvar(clientes.filter((c) => c.id !== id));
          setTela("dashboard");
        },
      },
    ]);
  }

  // ================= AUXILIAR: VERIFICA ATRASO =================
  function temParcelaAtrasada(cliente) {
    const hoje = new Date();
    hoje.setHours(0,0,0,0);

    if(!cliente.parcelasLista) return false;

    for (let p of cliente.parcelasLista) {
      if (p.pago < p.valor && p.dataVencimento) {
        const partes = p.dataVencimento.split("/");
        if (partes.length === 3) {
          const dataVenc = new Date(partes[2], partes[1] - 1, partes[0]);
          if (dataVenc < hoje) {
            return true; 
          }
        }
      }
    }
    return false;
  }

  // ================= TELA: NOVO CLIENTE =================

  if (tela === "novo") {
    return (
      <ScrollView style={styles.container}>
        <Text style={styles.title}>✨ Novo Cliente</Text>

        <TextInput style={styles.input} placeholder="Nome do Cliente" placeholderTextColor="#888" value={nome} onChangeText={setNome} />
        <TextInput style={styles.input} placeholder="Telefone (DDD + Número)" placeholderTextColor="#888" keyboardType="phone-pad" value={telefone} onChangeText={setTelefone} />
        <TextInput style={styles.input} placeholder="Valor Total do Crédito" placeholderTextColor="#888" keyboardType="numeric" value={valor} onChangeText={setValor} />
        <TextInput style={styles.input} placeholder="Quantidade de Parcelas" placeholderTextColor="#888" keyboardType="numeric" value={parcelas} onChangeText={setParcelas} />
        <TextInput style={styles.input} placeholder="Data do 1º Vencimento (DD/MM/AAAA)" placeholderTextColor="#888" value={dataPrimeiroVencimento} onChangeText={setDataPrimeiroVencimento} />

        <Text style={styles.labelForm}>Frequência de Cobrança:</Text>
        <View style={styles.rowFrequencia}>
          <TouchableOpacity style={[styles.selectorBtn, frequencia === "diario" && styles.selectorBtnAtivo]} onPress={() => setFrequencia("diario")}>
            <Text style={styles.btnText}>📅 Diário</Text>
          </TouchableOpacity>
          <TouchableOpacity style={[styles.selectorBtn, frequencia === "semanal" && styles.selectorBtnAtivo]} onPress={() => setFrequencia("semanal")}>
            <Text style={styles.btnText}>📆 Semanal</Text>
          </TouchableOpacity>
        </View>

        <TouchableOpacity style={styles.btn} onPress={adicionarCliente}>
          <Text style={styles.btnText}>💾 Salvar Cliente</Text>
        </TouchableOpacity>

        <TouchableOpacity onPress={() => setTela("dashboard")}>
          <Text style={styles.voltar}>← Voltar ao Dashboard</Text>
        </TouchableOpacity>
      </ScrollView>
    );
  }

  // ================= TELA: EDITAR CADASTRO =================

  if (tela === "editar") {
    return (
      <ScrollView style={styles.container}>
        <Text style={styles.title}>✏️ Editar Cadastro</Text>
        <Text style={styles.textGray}>Aviso: Alterar valor ou parcelas recriará o plano de cobrança.</Text>

        <TextInput style={styles.input} placeholder="Nome do Cliente" placeholderTextColor="#888" value={nome} onChangeText={setNome} />
        <TextInput style={styles.input} placeholder="Telefone" placeholderTextColor="#888" keyboardType="phone-pad" value={telefone} onChangeText={setTelefone} />
        <TextInput style={styles.input} placeholder="Valor Total" placeholderTextColor="#888" keyboardType="numeric" value={valor} onChangeText={setValor} />
        <TextInput style={styles.input} placeholder="Quantidade de Parcelas" placeholderTextColor="#888" keyboardType="numeric" value={parcelas} onChangeText={setParcelas} />
        <TextInput style={styles.input} placeholder="Data do 1º Vencimento (DD/MM/AAAA)" placeholderTextColor="#888" value={dataPrimeiroVencimento} onChangeText={setDataPrimeiroVencimento} />

        <Text style={styles.labelForm}>Frequência de Cobrança:</Text>
        <View style={styles.rowFrequencia}>
          <TouchableOpacity style={[styles.selectorBtn, frequencia === "diario" && styles.selectorBtnAtivo]} onPress={() => setFrequencia("diario")}>
            <Text style={styles.btnText}>📅 Diário</Text>
          </TouchableOpacity>
          <TouchableOpacity style={[styles.selectorBtn, frequencia === "semanal" && styles.selectorBtnAtivo]} onPress={() => setFrequencia("semanal")}>
            <Text style={styles.btnText}>📆 Semanal</Text>
          </TouchableOpacity>
        </View>

        <TouchableOpacity style={styles.btn} onPress={salvarEdicao}>
          <Text style={styles.btnText}>✅ Salvar Alterações</Text>
        </TouchableOpacity>

        <TouchableOpacity onPress={() => setTela("cliente")}>
          <Text style={styles.voltar}>❌ Cancelar</Text>
        </TouchableOpacity>
      </ScrollView>
    );
  }

  // ================= TELA: DETALHES DO CLIENTE =================

  if (tela === "cliente") {
    const c = clientes.find((x) => x.id === clienteAtual.id) || clienteAtual;
    const restante = c.valorTotal - c.pagoTotal;
    const atrasado = temParcelaAtrasada(c);

    return (
      <ScrollView style={styles.container}>
        <View style={styles.topBar}>
          <TouchableOpacity onPress={() => { setTela("dashboard"); setComprovante(""); }}>
            <Text style={styles.voltar}>← Voltar</Text>
          </TouchableOpacity>
          <View style={{ flexDirection: "row", gap: 15 }}>
            <TouchableOpacity onPress={() => abrirEditar(c)}>
              <Text style={styles.linkEditar}>✏️ Editar</Text>
            </TouchableOpacity>
            <TouchableOpacity onPress={() => excluirCliente(c.id)}>
              <Text style={styles.lixeira}>🗑️ Excluir</Text>
            </TouchableOpacity>
          </View>
        </View>

        <View style={styles.clienteHeaderBox}>
          <Text style={styles.title}>{c.nome}</Text>
          {atrasado && (
            <View style={styles.badgeAlerta}>
              <Text style={styles.badgeAlertaTexto}>⚠️ PARCELA ATRASADA</Text>
            </View>
          )}
        </View>
        <Text style={styles.phoneText}>📞 {c.telefone || "Sem telefone"} | Tipo: {c.frequencia === "semanal" ? "📆 Semanal" : "📅 Diário"}</Text>

        {/* Resumo Financeiro */}
        <View style={styles.resumoClienteBox}>
          <View style={styles.resumoLinha}><Text style={styles.textGray}>Total Financiado:</Text><Text style={styles.textBranco}>R$ {c.valorTotal.toFixed(2)}</Text></View>
          <View style={styles.resumoLinha}><Text style={styles.textGray}>Total Pago:</Text><Text style={styles.verde}>R$ {c.pagoTotal.toFixed(2)}</Text></View>
          <View style={styles.resumoLinha}><Text style={styles.textGray}>Saldo Restante:</Text><Text style={restante > 0 && atrasado ? styles.vermelho : styles.laranja}>R$ {restante.toFixed(2)}</Text></View>
        </View>

        {/* Botão de Cobrança Rápida */}
        {restante > 0 && (
          <TouchableOpacity style={styles.btnCobranca} onPress={() => enviarCobranca(c)}>
            <Text style={styles.btnText}>📲 Enviar Lembrete de Cobrança</Text>
          </TouchableOpacity>
        )}

        {/* Receber Pagamento */}
        <TextInput 
          style={styles.input} 
          placeholder="Valor a pagar (R$)" 
          placeholderTextColor="#888" 
          keyboardType="numeric"
          value={valorPagamento} 
          onChangeText={setValorPagamento} 
        />

        <TouchableOpacity style={styles.btn} onPress={() => { pagarManual(clientes.indexOf(c), valorPagamento); setValorPagamento(""); }}>
          <Text style={styles.btnText}>💵 Receber Pagamento</Text>
        </TouchableOpacity>

        <TouchableOpacity style={styles.btnRemover} onPress={() => removerUltimoPagamento(clientes.indexOf(c))}>
          <Text style={styles.btnText}>↩️ Estornar Último Pagamento</Text>
        </TouchableOpacity>

        {/* Bloco de Comprovante */}
        {comprovante !== "" && (
          <View style={styles.comprovanteBox}>
            <Text style={styles.comprovanteTitulo}>✅ Pagamento Registrado!</Text>
            <Text style={styles.comprovanteTexto}>{comprovante}</Text>
            <TouchableOpacity style={styles.btnWhats} onPress={enviarWhatsApp}>
              <Text style={styles.btnText}>💬 Enviar via WhatsApp</Text>
            </TouchableOpacity>
          </View>
        )}

        <Text style={styles.subtitle}>📊 Carnê de Parcelas</Text>
        {c.parcelasLista && c.parcelasLista.map((p, i) => {
          const isPaga = p.pago >= p.valor;
          
          let isVencida = false;
          if (!isPaga && p.dataVencimento) {
            const partes = p.dataVencimento.split("/");
            if (partes.length === 3) {
              const dataVenc = new Date(partes[2], partes[1] - 1, partes[0]);
              const hoje = new Date(); hoje.setHours(0,0,0,0);
              isVencida = dataVenc < hoje;
            }
          }

          return (
            <View key={i} style={[styles.parcelaBox, isPaga && styles.parcelaPagaBg, isVencida && styles.parcelaVencidaBg]}>
              <View style={{ flexDirection: "row", justifyContent: "space-between" }}>
                <Text style={[styles.text, isPaga && styles.verde, isVencida && styles.vermelhoTexto]}>
                  Parcela {p.numero} - R$ {p.valor.toFixed(2)}
                </Text>
                <Text style={isVencida ? styles.vermelhoTexto : styles.textGray}>
                  {p.dataVencimento ? `Venc: ${p.dataVencimento}` : "Sem Vencimento"}
                </Text>
              </View>
              {isPaga && <Text style={styles.dataPagaText}>📅 Recebido em: {p.dataPagamento}</Text>}
              {!isPaga && p.pago > 0 && <Text style={styles.laranja}>Abatido: R$ {p.pago.toFixed(2)} (Falta R$ {(p.valor - p.pago).toFixed(2)})</Text>}
            </View>
          );
        })}
        <View style={{ height: 40 }} />
      </ScrollView>
    );
  }

  // ================= TELA: DASHBOARD =================

  const totalRestante = clientes.reduce((t, c) => t + (c.valorTotal - c.pagoTotal), 0);

  const clientesFiltrados = clientes.filter(c => 
    c.nome.toLowerCase().includes(pesquisa.toLowerCase())
  );

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.titleHeader}>Seven Crédito</Text>
      
      {/* Cards Financeiros */}
      <View style={styles.dashboardGrid}>
        <View style={styles.miniCard}>
          <Text style={styles.cardLabel}>Caixa Geral</Text>
          <Text style={styles.cardValue}>R$ {caixa.toFixed(2)}</Text>
        </View>
        <View style={styles.miniCard}>
          <Text style={styles.cardLabel}>Entradas do Dia</Text>
          <Text style={styles.cardValueGreen}>R$ {entradaDia.toFixed(2)}</Text>
        </View>
      </View>

      <View style={[styles.miniCard, { width: "100%", marginTop: 10 }]}>
        <Text style={styles.cardLabel}>Total a Receber no Mercado</Text>
        <Text style={styles.cardValueLaranja}>R$ {totalRestante.toFixed(2)}</Text>
      </View>

      {/* Ajuste de Caixa */}
      <View style={styles.secaoCaixa}>
        <Text style={styles.subtitle}>⚙️ Ajuste Rápido de Caixa</Text>
        <View style={styles.caixaInputRow}>
          <TextInput
            placeholder="Ex: 50 ou -30"
            placeholderTextColor="#888"
            keyboardType="numeric"
            style={[styles.input, { flex: 1, marginTop: 0 }]}
            value={valorCaixa}
            onChangeText={setValorCaixa}
          />
          <TouchableOpacity style={styles.btnAjuste} onPress={ajustarCaixa}>
            <Text style={styles.btnText}>Ajustar</Text>
          </TouchableOpacity>
        </View>
      </View>

      {/* CABEÇALHO CLIENTES + BUSCA */}
      <View style={styles.clientesHeaderRow}>
        <Text style={styles.subtitle}>👥 Clientes Ativos ({clientes.length})</Text>
        <TouchableOpacity style={styles.btnNovoCliente} onPress={() => {
          setDataPrimeiroVencimento(new Date().toLocaleDateString("pt-BR"));
          setTela("novo");
        }}>
          <Text style={styles.btnText}>+ Novo</Text>
        </TouchableOpacity>
      </View>

      {/* Barra de Busca */}
      <TextInput
        style={styles.searchBar}
        placeholder="🔍 Buscar cliente pelo nome..."
        placeholderTextColor="#888"
        value={pesquisa}
        onChangeText={setPesquisa}
      />

      {clientesFiltrados.length === 0 ? (
        <Text style={styles.emptyText}>
          {clientes.length === 0 ? "Nenhum cliente cadastrado." : "Nenhum cliente encontrado na busca."}
        </Text>
      ) : (
        clientesFiltrados.map((c, i) => {
          const devedor = c.valorTotal - c.pagoTotal;
          const emAtraso = temParcelaAtrasada(c);

          return (
            <TouchableOpacity key={c.id} style={[styles.card, emAtraso && styles.cardBordaAtraso]} onPress={() => { setClienteAtual(c); setTela("cliente"); }}>
              <View style={styles.cardInfo}>
                <Text style={styles.nome}>{c.nome}</Text>
                <Text style={styles.subnome}>Restam: R$ {devedor.toFixed(2)} | {c.frequencia === "semanal" ? "Semanal" : "Diário"}</Text>
              </View>
              <View style={styles.badgeStatus}>
                <Text style={devedor === 0 ? styles.statusQuitado : emAtraso ? styles.statusAtrasado : styles.statusPendente}>
                  {devedor === 0 ? "QUITADO" : emAtraso ? "⚠️ ATRASADO" : "EM DIA"}
                </Text>
              </View>
            </TouchableOpacity>
          );
        })
      )}
      <View style={{ height: 40 }} />
    </ScrollView>
  );
}
